  kubectl -n {{ .Release.Namespace }} get kasprapp -w

The control plane's API, from your machine:

  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "midstream.controlPlane" . }}-app-api {{ .Values.controlPlane.webPort }}:{{ .Values.controlPlane.webPort }}
  curl -s localhost:{{ .Values.controlPlane.webPort }}/system/version

{{- end }}
